1# -*- coding: utf-8 -*- 

2"""Location: ./mcpgateway/routers/search.py 

3Copyright contributors to the MCP-CONTEXT-FORGE project 

4SPDX-License-Identifier: Apache-2.0 

5 

6Unified Search API Router. 

7 

8Exposes unified cross-entity search at a stable, versioned, non-admin path 

9(``GET /v1/search``) so client-facing callers do not depend on 

10``GET /admin/search``, which is gated on ``admin.dashboard`` and slated for 

11deprecation. 

12 

13Security model: 

14 Authentication only at the top level (no ``admin.dashboard`` gate). Real 

15 authorization is enforced per-entity inside 

16 :func:`mcpgateway.admin.perform_unified_search`, and token scoping still 

17 filters visible entities. 

18""" 

19 

20# Standard 

21from typing import Any, Optional

23# Third-Party 

24from fastapi import APIRouter, Depends, Query, Request 

25from fastapi.responses import JSONResponse 

26from sqlalchemy.orm import Session 

27 

28# First-Party 

29from mcpgateway.common.query_params import QueryEntityTypes, QueryGatewayIdList, QueryTagsFilter 

30from mcpgateway.config import settings 

31from mcpgateway.db import get_db 

32from mcpgateway.middleware.rbac import get_current_user_with_permissions 

33

56 """Unified search across primary entities (versioned, non-admin route). 

57 

58 Same behavior and response shape as ``GET /admin/search`` without the 

59 ``admin.dashboard`` gate; delegates to 

60 :func:`mcpgateway.admin.perform_unified_search`. 

61 

62 Args: 

63 request: Current request object. 

64 q (str): Free-text search query. 

65 tags (Optional[str]): Tag filter expression (comma=OR, plus=AND). 

66 entity_types (Optional[str]): Optional comma-separated entity type list. 

67 Supported values: servers, gateways, tools, resources, prompts, 

68 agents, teams, users, roots. 

69 include_inactive (bool): Whether to include inactive entities. 

70 limit (int): Default per-entity limit for returned items. 

71 limit_per_type (Optional[int]): Optional alias overriding ``limit``. 

72 gateway_id (Optional[str]): Gateway filter for tools/resources/prompts. 

73 team_id (Optional[str]): Team scope filter. 

74 db (Session): Database session. 

75 user: Authenticated user context. 

76 

77 Returns: 

78 dict[str, Any]: Grouped and flattened search results with metadata.
