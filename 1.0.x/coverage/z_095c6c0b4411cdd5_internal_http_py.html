22def _is_ssl_enabled() -> bool: 

23 """Check whether the gateway is running with SSL enabled. 

24

31def internal_loopback_base_url() -> str: 

32 """Return loopback base URL for gateway self-calls. 

33 

34 Uses HTTPS when runtime is started with SSL=true, otherwise HTTP. 

35 

36 Returns: 

37 str: The base URL string (e.g. ``http://127.0.0.1:4444``).

39 scheme = "https" if _is_ssl_enabled() else "http" 

40 return f"{scheme}://127.0.0.1:{settings.port}"

43def internal_loopback_verify() -> bool: 

44 """Return TLS verification policy for loopback self-calls. 

45 

46 Loopback HTTPS frequently uses a self-signed local cert, so verification 

47 is disabled for HTTPS loopback self-calls and enabled otherwise. 

48 

49 Returns: 

50 bool: ``False`` when the loopback URL is HTTPS, ``True`` otherwise.

52 return not _is_ssl_enabled()

55async def post_rpc_in_process(*, content: bytes, headers: dict, timeout: float, auth_context: str) -> httpx.Response: 

56 """POST to the trusted-internal ``/_internal/mcp/rpc`` route via an in-process ASGI transport. 

57 

58 Affinity-owned requests must execute on the worker that actually holds the 

59 bound upstream session. A real loopback to ``127.0.0.1`` hits the shared 

60 gunicorn socket, and the kernel routes it to an arbitrary worker that does 

61 not hold the session, which breaks upstream-session reuse (and the #4205 

62 isolation invariant for stateful upstreams). ``httpx.ASGITransport`` invokes 

63 the FastAPI app in *this* process instead, so the dispatch resolves the 

64 session from this worker's ``UpstreamSessionRegistry``. 

65 

66 Targets the trusted-internal endpoint rather than the public ``/rpc`` so that 

67 OAuth and ``MCP_REQUIRE_AUTH=false`` public-only sessions are not 

68 re-authenticated (and 401'd) at the public route boundary. The already 

69 validated edge identity rides in ``auth_context``; this helper attaches the 

70 runtime-auth trust headers and the encoded context, and pins the ASGI client 

71 to loopback so the trust gate accepts the call. It is trust-agnostic: a 

72 Redis-forwarded request MUST have had its signature verified by the caller 

73 before reaching here. 

74 

75 ``app`` is imported lazily to avoid a circular import at module load. 

76 

77 Args: 

78 content: Serialized JSON-RPC request body. 

79 headers: Request headers. Must include ``x-forwarded-internally: true`` 

80 so the re-entered handler does not forward again. 

81 timeout: Per-call timeout in seconds. 

82 auth_context: Encoded ``x-contextforge-auth-context`` value (required, 

83 non-empty) carrying the edge-validated identity. Attached verbatim; 

84 this helper does not verify it. 

85 

86 Returns: 

87 httpx.Response: The response from the in-process ``/_internal/mcp/rpc`` dispatch. 

88 

89 Raises: 

90 ValueError: If ``auth_context`` is empty (internal dispatch must always carry one). 

91 """ 

92 if not auth_context: 

93 raise ValueError("post_rpc_in_process requires a non-empty auth_context for trusted-internal dispatch") 

94

96 from mcpgateway.auth_context import _expected_internal_mcp_runtime_auth_header # pylint: disable=import-outside-toplevel,protected-access 

97 from mcpgateway.main import app # pylint: disable=import-outside-toplevel,cyclic-import 

98 

99 # Trust headers for the internal endpoint: the "affinity" runtime marker, the 

100 # shared-secret HMAC, and the encoded edge auth context (so the endpoint 

101 # reconstructs the caller without re-authenticating). 

102 rpc_headers = dict(headers) 

103 rpc_headers["x-contextforge-mcp-runtime"] = "affinity" 

104 rpc_headers["x-contextforge-mcp-runtime-auth"] = _expected_internal_mcp_runtime_auth_header() 

105 rpc_headers["x-contextforge-auth-context"] = auth_context 

106 

107 # client=("127.0.0.1", 0) sets scope["client"] to a loopback address so the 

108 # trust gate's defense-in-depth loopback check accepts the in-process call. 

109 transport = httpx.ASGITransport(app=app, client=("127.0.0.1", 0)) 

110 async with httpx.AsyncClient(transport=transport, base_url=internal_loopback_base_url()) as client: 

111 return await client.post("/_internal/mcp/rpc", content=content, headers=rpc_headers, timeout=timeout)
