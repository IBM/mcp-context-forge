2"""Location: ./mcpgateway/utils/oauth_resource.py 

3Copyright contributors to the MCP-CONTEXT-FORGE project 

4SPDX-License-Identifier: Apache-2.0

6RFC 8707 Resource Indicator utilities. 

7 

8Single home for everything ``resource``-shaped in the OAuth flows: 

9 

10* :func:`derive_resource_origin` — the auto-derived audience heuristic 

11 (``scheme://netloc`` of the upstream MCP server URL), used identically by 

12 the authorization request, the callback token exchange, the refresh 

13 request, and the inbound validator fallback so the IdP always sees one 

14 consistent ``resource`` value for a given gateway. 

15* :func:`normalize_resource` — RFC 8707 canonicalization for outbound 

16 ``resource`` parameters (fragment stripped; query policy by caller). 

17* :func:`parse_oauth_resource_form` — Admin UI form input → single URI or 

18 list of URIs, preserving the string/list shapes IdPs use for ``aud`` 

19 (RFC 7519 §4.1.3). 

20 

21Note on standards status: RFC 8707 §2.1 does not specify deriving a default 

22``resource`` from the target URL when the client omits one — the origin 

23derivation here is a deliberate implementation heuristic that matches the 

24MCP spec's canonical-URI convention and the origin-level audiences issued 

25by major IdPs (Salesforce, Azure AD, Okta). It is not an RFC-mandated 

26behavior; configured or learned ``resource`` values always take precedence. 

27 

28Examples: 

29 >>> derive_resource_origin("https://api.salesforce.com/platform/mcp/v1/x") 

30 'https://api.salesforce.com' 

31 >>> derive_resource_origin("urn:example:app") is None 

32 True 

33 >>> normalize_resource("https://api.example.com/mcp?x=1#frag") 

34 'https://api.example.com/mcp' 

35 >>> normalize_resource("https://api.example.com/mcp?x=1", preserve_query=True) 

36 'https://api.example.com/mcp?x=1' 

37 >>> normalize_resource("opaque-client-id") 

38 'opaque-client-id' 

39 >>> parse_oauth_resource_form("https://a.example.com, https://b.example.com") 

40 ['https://a.example.com', 'https://b.example.com'] 

41 >>> parse_oauth_resource_form("https://api.example.com/path?x=1,y=2") 

42 'https://api.example.com/path?x=1,y=2' 

43"""

54def derive_resource_origin(url: Optional[str]) -> Optional[str]: 

55 """Derive the origin (scheme + netloc) from a gateway URL for use as a fallback resource. 

56 

57 Real-world OAuth providers (Salesforce, Azure AD, Okta) issue access tokens 

58 with origin-level audiences (``https://api.salesforce.com``) rather than the 

59 full URL of the protected MCP endpoint 

60 (``https://api.salesforce.com/platform/mcp/v1/...``). Using the full 

61 gateway URL as the auto-derived resource therefore reliably mismatches the 

62 token's actual aud and produces validation failures. 

63 

64 This helper is the single source of truth for that derivation. It is used 

65 by three call sites that must agree on the auto-derived audience: 

66 

67 * ``mcpgateway.routers.oauth_router`` — outbound ``resource`` param on the 

68 initial authorization request and the callback token exchange. 

69 * ``mcpgateway.services.token_storage_service`` — outbound ``resource`` 

70 param on token refresh (must match the initial request so the IdP mints 

71 a token for the same audience). 

72 * ``mcpgateway.services.token_validation_service`` — inbound ``aud`` 

73 comparison fallback when no admin-configured or per-user-learned 

74 resource exists. 

75 

76 Args: 

77 url: Gateway URL to extract the origin from. 

78 

79 Returns: 

80 ``scheme://netloc`` for hierarchical URLs, or ``None`` for URNs / empty 

81 / scheme-less inputs (caller should treat as "no auto-fallback 

82 possible" and rely on admin config or the per-user learned value). 

83 

84 Examples: 

85 >>> derive_resource_origin("https://api.example.com/mcp/v1?x=1#f") 

86 'https://api.example.com' 

87 >>> derive_resource_origin("http://localhost:9000/sse") 

88 'http://localhost:9000' 

89 >>> derive_resource_origin("") is None 

90 True 

91 >>> derive_resource_origin(None) is None 

92 True 

93 >>> derive_resource_origin("urn:example:app") is None 

94 True 

95 >>> derive_resource_origin("/relative/path") is None 

96 True 

97 """ 

98 if not url: 

99 return None 

100 parsed = urlparse(url) 

101 if not parsed.scheme or not parsed.netloc: 

102 return None 

103 return f"{parsed.scheme}://{parsed.netloc}"

106def normalize_resource(url: Optional[str], *, preserve_query: bool = False) -> Optional[str]: 

107 """Normalize a resource value per RFC 8707, or pass through opaque identifiers. 

108 

109 URL-shaped inputs are canonicalized (fragment stripped — RFC 8707 §2 

110 MUST NOT; query stripped unless ``preserve_query`` — §2 SHOULD NOT). 

111 Non-URL inputs (no scheme) are returned verbatim so that opaque audience 

112 identifiers learned from IdPs that do not honor RFC 8707 (e.g. ServiceNow 

113 / Authentik returning ``aud=client_id``) round-trip correctly. RFC 8707 

114 §2 explicitly permits the AS to map ``resource`` to an abstract 

115 identifier; the resource server therefore must accept either form. 

116 

117 Args: 

118 url: Resource URL or opaque audience identifier to normalize. 

119 preserve_query: If True, preserve query (for explicitly configured 

120 resources, where the query may be significant). If False, strip 

121 it (RFC 8707 SHOULD NOT for auto-derived values). 

122 

123 Returns: 

124 Normalized URL string, the original opaque value, or ``None`` if the 

125 input is empty. 

126 

127 Examples: 

128 >>> normalize_resource("https://api.example.com/mcp#section") 

129 'https://api.example.com/mcp' 

130 >>> normalize_resource("https://api.example.com/mcp?x=1") 

131 'https://api.example.com/mcp' 

132 >>> normalize_resource("https://api.example.com/mcp?x=1", preserve_query=True) 

133 'https://api.example.com/mcp?x=1' 

134 >>> normalize_resource("test-servicenow-opaque-client-id") 

135 'test-servicenow-opaque-client-id' 

136 >>> normalize_resource("") is None 

137 True 

138 """ 

139 if not url: 

140 return None 

141 parsed = urlparse(url) 

142 # If the value lacks a scheme it is not a URL; treat as an opaque 

143 # audience identifier and pass through verbatim so a learned 

144 # client_id-style audience survives refresh. 

145 if not parsed.scheme: 

146 return url 

147 # Remove fragment (MUST NOT); query: preserve for explicit, strip for auto-derived 

148 query = parsed.query if preserve_query else "" 

149 return urlunparse((parsed.scheme, parsed.netloc, parsed.path, parsed.params, query, ""))

152def parse_oauth_resource_form(raw: Any) -> Union[str, List[str], None]: 

153 """Parse the ``oauth_resource`` Admin UI form field into a single URI or list of URIs. 

154 

155 Splits on whitespace, newlines, or commas, but only accepts the multi-value 

156 interpretation when every resulting piece is a well-formed absolute URI (has 

157 a scheme). This preserves single resource URIs that legitimately contain 

158 commas in their path or query component (RFC 3986 pchar) rather than 

159 silently corrupting them into two bogus entries. 

160 

161 Storage convention: single value is stored as ``str``, multiple as 

162 ``list[str]``, matching the shapes that IdPs use for the ``aud`` claim 

163 (RFC 7519 §4.1.3) and that :func:`mcpgateway.services.token_validation_service` 

164 accepts for ``oauth_config["resource"]``. 

165 

166 Args: 

167 raw: Value pulled from a form field (typically ``form.get("oauth_resource")``). 

168 Non-string values, empty strings, and whitespace-only strings all 

169 yield ``None``. 

170 

171 Returns: 

172 ``None`` for empty input, a ``str`` for a single resource, or a 

173 ``list[str]`` for multiple. 

174 

175 Examples: 

176 >>> parse_oauth_resource_form(None) is None 

177 True 

178 >>> parse_oauth_resource_form("") is None 

179 True 

180 >>> parse_oauth_resource_form(" ") is None 

181 True 

182 >>> parse_oauth_resource_form("https://api.example.com") 

183 'https://api.example.com' 

184 >>> parse_oauth_resource_form("https://a.example.com, https://b.example.com") 

185 ['https://a.example.com', 'https://b.example.com'] 

186 >>> parse_oauth_resource_form("https://a.example.com\\nhttps://b.example.com") 

187 ['https://a.example.com', 'https://b.example.com'] 

188 >>> parse_oauth_resource_form("https://api.example.com/path?x=1,y=2") 

189 'https://api.example.com/path?x=1,y=2' 

190 """ 

191 if not isinstance(raw, str): 

192 return None 

193 stripped = raw.strip() 

194 if not stripped: 

195 return None 

196 pieces = [p.strip() for p in re.split(r"[,\s]+", stripped) if p.strip()] 

197 if not pieces: 

198 return None 

199 if len(pieces) == 1: 

200 return pieces[0] 

201 # Multi-value only accepted when every piece parses as an absolute URI. 

202 # Falling back to single-value protects URIs containing unencoded commas 

203 # (RFC 3986 pchar allows ',' in path and query components). 

204 if all(urlparse(p).scheme for p in pieces): 

205 return pieces 

206 return stripped
