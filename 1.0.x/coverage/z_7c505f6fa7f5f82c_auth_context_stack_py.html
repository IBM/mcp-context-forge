1# -*- coding: utf-8 -*- 

2"""Location: ./mcpgateway/middleware/auth_context_stack.py 

3Copyright contributors to the MCP-CONTEXT-FORGE project 

4SPDX-License-Identifier: Apache-2.0 

5 

6Registration helper for the CSRF / password-change-enforcement / 

7auth-context middleware stack. 

8 

9Extracted from ``mcpgateway.main`` so the registration order can be 

10unit-tested against a bare FastAPI app without importing the assembled 

11application singleton (whose middleware composition depends on the settings 

12in effect at first import). 

13""" 

14 

15# Standard 

16import logging 

17 

18# Third-Party 

19from fastapi import FastAPI 

20 

21# First-Party 

22from mcpgateway.config import settings 

23 

24logger = logging.getLogger(__name__) 

25 

26 

27def register_auth_context_middleware(app: FastAPI) -> None: 

28 """Register CSRF, password-change-enforcement, and auth-context middleware. 

29 

30 Registration order is load-bearing (issue #5739): Starlette executes user 

31 middleware in reverse registration order, so CSRFMiddleware is registered 

32 first and AuthContextMiddleware last, meaning AuthContextMiddleware runs 

33 first and populates ``request.state.user`` before CSRFMiddleware resolves 

34 its identity. Getting this backwards makes CSRFMiddleware silently fall 

35 back to resolving identity from the raw JWT ``sub`` claim (EmailUser.id) 

36 instead of the email admin.py binds CSRF tokens to. 

37 

38 Args: 

39 app: The FastAPI application to register the middleware on. 

40 """ 

41 # Add authentication context middleware if security logging is enabled OR password change enforcement is enabled 

42 # This middleware extracts user context and logs security events (authentication attempts) 

43 # Note: SIEM export can also require auth event capture even when DB security logging is off. 

44 # Note: Password change enforcement also requires user context to be available 

45 siem_auth_source_enabled = settings.siem_export_enabled and "auth" in {str(item).lower() for item in getattr(settings, "siem_export_event_sources", [])} 

46 

47 # Add CSRF protection middleware FIRST (runs LAST/innermost due to reverse order) 

48 # This validates CSRF tokens on state-changing requests to prevent Cross-Site Request Forgery attacks 

49 # Must be added before AuthContextMiddleware below so it executes AFTER it and request.state.user 

50 # is available for token validation (adding it later would make it run BEFORE AuthContextMiddleware, 

51 # leaving request.state.user unset and silently falling back to resolving identity from the raw JWT 

52 # `sub` claim (EmailUser.id) instead of the email admin.py binds CSRF tokens to). 

53 if settings.csrf_enabled: 

54 # First-Party 

55 from mcpgateway.middleware.csrf_middleware import CSRFMiddleware 

56 

57 app.add_middleware(CSRFMiddleware) 

58 logger.info("🛡️ CSRF protection middleware enabled - validating tokens on state-changing requests") 

59 else: 

60 logger.info("🛡️ CSRF protection middleware disabled") 

61 

62 # Add password change enforcement middleware FIRST (runs SECOND due to reverse order) 

63 # This middleware enforces mandatory password changes for users with password_change_required flag 

64 # Note: Runs after AuthContextMiddleware (added below) so request.state.user is available 

65 if settings.password_change_enforcement_enabled: 

66 # First-Party 

67 from mcpgateway.middleware.password_change_enforcement import PasswordChangeEnforcementMiddleware 

68 

69 app.add_middleware(PasswordChangeEnforcementMiddleware) 

70 logger.info("🔒 Password change enforcement middleware enabled - blocking access for users requiring password change") 

71 else: 

72 logger.info("🔒 Password change enforcement middleware disabled") 

73 

74 # Add authentication context middleware SECOND (runs FIRST due to reverse order) 

75 # This populates request.state.user for downstream middleware and handlers 

76 auth_context_required = settings.security_logging_enabled or siem_auth_source_enabled or settings.mcpgateway_admin_api_enabled or settings.password_change_enforcement_enabled 

77 if auth_context_required: 

78 # First-Party 

79 from mcpgateway.middleware.auth_middleware import AuthContextMiddleware 

80 

81 app.add_middleware(AuthContextMiddleware) 

82 logger.info("🔐 Authentication context middleware enabled - capturing authentication security events") 

83 else: 

84 logger.info("🔐 Security event logging disabled")
