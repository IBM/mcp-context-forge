21# --------------------------------------------------------------------------- 

22# G1: plugin result.metadata -> observability consumption 

23# --------------------------------------------------------------------------- 

24# 

25# S4 bounds (untrusted plugin output). These are deliberately conservative, 

26# defensible defaults -- not a contract mandated by any single plugin: 

27# - plugin_name and metric keys must match _IDENTIFIER_RE (bounded length, 

28# restricted charset) before they're allowed to flow into DB span/metric 

29# names, resource ids, or OTel span/attribute names -- these are 

30# plugin-controlled and otherwise unvalidated. 

31# - Only scalar values (str, int, float, bool) are recorded as-is; strings 

32# must additionally have a field name in the explicit _SAFE_STRING_FIELD_NAMES 

33# allowlist and be within _MAX_STRING_LENGTH and match _SAFE_STRING_VALUE_RE 

34# (a low-cardinality, no-free-text charset). The charset alone bounds size and 

35# character set, not semantic sensitivity -- a short SSN- or token-shaped value 

36# (e.g. "123-45-6789") is charset-valid, so string values are deny-by-default 

37# and only accepted for field names known in advance to carry non-sensitive 

38# enum/status/type data. Overlong values are rejected outright, not truncated -- 

39# truncation alone would bound size but not the sensitivity of what's exported 

40# (e.g. a truncated secret/token fragment is still a secret/token fragment). 

41# - Numeric (int/float, excluding bool) values are subject to the same 

42# deny-by-default field-name allowlist (_SAFE_NUMERIC_FIELD_NAMES): a field 

43# name valid as an identifier is not proof it's non-sensitive -- {"ssn": 

44# 123456789} or {"account_id": 123456789012} would otherwise sail through 

45# under any plugin-chosen key. Only fields known in advance to carry 

46# non-sensitive counters (e.g. total_detections, total_masked) are accepted, 

47# and the value must additionally be a finite number (NaN/inf rejected). 

48# - A list[str] value (e.g. ["email", "ssn"]) is special-cased: joined into 

49# a single comma-separated string (OTel span attributes don't nest), then 

50# subject to the same field-name-allowlisted string-value validation as any 

51# other string value. 

52# - Any other type (dict, mixed/non-str list, None, etc.) is dropped. 

53# - At most _MAX_PLUGIN_KEYS keys survive per plugin's metadata dict, and 

54# only the first _MAX_PLUGIN_KEYS items of the raw dict are ever inspected 

55# (via itertools.islice) -- not just accepted -- so a huge untrusted dict, 

56# even one front-loaded with invalid keys, can't force a full scan. 

57# - At most _MAX_PLUGINS_PER_CALL plugin namespaces are processed per call 

58# (result.metadata can in principle carry entries from every plugin that 

59# ran in the hook chain); the cap is applied before the full item list is 

60# materialized. 

61# - list[str] values are only checked for str-ness and per-item length 

62# within the first _MAX_LIST_ITEMS items, so an oversized untrusted list 

63# can't force a full scan or an unbounded join before it's bounded. 

64# On drop, only the key name and a short reason are logged -- never the value 

65# itself, since it may be malformed/oversized/adversarial plugin output. 

66_MAX_PLUGIN_KEYS = 32 

67_MAX_STRING_LENGTH = 64 

68_MAX_PLUGINS_PER_CALL = 16 

69_MAX_LIST_ITEMS = 32 

70 

71# Identifier contract for plugin_name / metric field names: these flow into DB 

72# span/metric names, resource ids, and OTel span/attribute names, so they're 

73# restricted to a short, safe charset rather than trusted as-is. 

74_IDENTIFIER_RE = re.compile(r"^[A-Za-z0-9_.-]{1,64}$") 

75 

76# Low-cardinality value contract for string metadata values: truncation alone 

77# bounds size, not sensitivity, so free text (arbitrary PII, spaces, quotes, 

78# etc.) is rejected outright rather than truncated. This intentionally only 

79# admits short enum/status/type-name-like tokens. 

80_SAFE_STRING_VALUE_RE = re.compile(r"^[A-Za-z0-9_.,:=/-]*$") 

81 

82# Deny-by-default field-name allowlist for string (and joined list[str]) values: the 

83# charset above bounds size and character set, but not semantic sensitivity -- a short 

84# SSN- or token-shaped value (e.g. "123-45-6789") is charset-valid and would otherwise 

85# sail through untouched. A string value is therefore only accepted when its field name 

86# is already known to carry non-sensitive enum/status/type data; every other field name 

87# carrying a string is dropped regardless of its content. New plugins that add a 

88# string-valued metadata field must extend this set explicitly -- see 

89# https://github.com/IBM/mcp-context-forge/issues/5554 and 

90# https://github.com/IBM/cpex-plugins/issues/129 (secrets_detection: secret_types, 

91# encoded_exfil_detection: encoding_types, url_reputation: reputation_categories, 

92# rate_limiter: backend -- wired up here; retry_with_backoff has no string field). 

93_SAFE_STRING_FIELD_NAMES = frozenset({"stage", "detection_types", "secret_types", "encoding_types", "reputation_categories", "backend"}) 

94 

95# Deny-by-default field-name allowlist for numeric (int/float, excluding bool) values: 

96# a field name matching _IDENTIFIER_RE is bounded/charset-safe but not proof the *value* 

97# is non-sensitive -- a plugin could report {"ssn": 123456789} or {"account_id": 

98# 123456789012} under any identifier it chooses, and that value would otherwise flow 

99# unchecked into DB span attributes, metric rows, and the OTel export sink. A numeric 

100# value is therefore only accepted when its field name is already known to carry a 

101# non-sensitive counter; every other field name carrying a number is dropped regardless 

102# of its value. New plugins that add a numeric-valued metadata field must extend this 

103# set explicitly -- see https://github.com/IBM/mcp-context-forge/issues/5554 and 

104# https://github.com/IBM/cpex-plugins/issues/129 (secrets_detection: total_blocked, 

105# url_reputation: total_checked, rate_limiter: allowed/throttled, retry_with_backoff: 

106# retry_count/retry_delay_ms -- wired up here; total_detections/total_masked above 

107# already cover secrets_detection's and encoded_exfil_detection's shared field names). 

108_SAFE_NUMERIC_FIELD_NAMES = frozenset({"total_detections", "total_masked", "total_blocked", "total_checked", "allowed", "throttled", "retry_count", "retry_delay_ms"})

111def _is_valid_identifier(name: Any) -> bool: 

112 """Check whether a plugin name or metric field name is safe to use as an identifier. 

113 

114 Args: 

115 name: Candidate identifier (plugin name or metric field name); treated as untrusted. 

116 

117 Returns: 

118 True if ``name`` is a string matching the bounded, restricted-charset 

119 identifier contract; False otherwise. 

120 """ 

121 return isinstance(name, str) and bool(_IDENTIFIER_RE.match(name))

124def _validate_string_value(value: str) -> Optional[str]: 

125 """Validate a single string metadata value (S4). 

126 

127 Args: 

128 value: Candidate string value. 

129 

130 Returns: 

131 ``value`` unchanged if it is within ``_MAX_STRING_LENGTH`` and matches 

132 the low-cardinality safe-value charset, otherwise ``None`` (reject). 

133 Overlong values are rejected outright rather than truncated -- a 

134 truncated secret/token/hash fragment is still sensitive, so length is 

135 checked before the charset is even considered. 

136 """ 

137 if len(value) > _MAX_STRING_LENGTH: 

138 return None 

139 if not _SAFE_STRING_VALUE_RE.match(value): 

140 return None 

141 return value

144def _validate_numeric_value(value: Any) -> Optional[Any]: 

145 """Validate a single numeric metadata value (S4). 

146 

147 Args: 

148 value: Candidate ``int``/``float`` value (never ``bool`` -- callers check that first). 

149 

150 Returns: 

151 ``value`` unchanged if it is a finite number, otherwise ``None`` (reject). 

152 ``NaN``/``inf``/``-inf`` are rejected outright since they can't be compared or 

153 aggregated meaningfully once exported as a metric/span attribute. Arbitrary- 

154 precision Python ints too large to convert to ``float`` (e.g. ``10**100000``) 

155 raise ``OverflowError`` from ``math.isfinite()`` -- also rejected, since the 

156 caller is untrusted plugin metadata and one such value must not abort 

157 validation of the rest of the hook's metrics. 

158 """ 

159 try: 

160 if not math.isfinite(value): 

161 return None 

162 except (OverflowError, TypeError): 

163 return None 

164 return value

167def _sanitize_plugin_metrics(plugin_name: str, raw_metrics: Any) -> Dict[str, Any]: 

168 """Validate and bound a single plugin's ``result.metadata[<plugin>]`` dict (S4). 

169 

170 Untrusted plugin output: only scalar (``str``/``int``/``float``/``bool``) values 

171 survive, and only when explicitly allowlisted -- ``bool`` is always accepted as-is; 

172 ``int``/``float`` and ``str`` (plus joined ``list[str]``) each require the field name 

173 to be in their respective deny-by-default allowlist. Everything else is dropped. Key 

174 names, key count, numeric values, and string values are all validated/capped. Dropped 

175 values are never logged -- only the key name and a short reason. 

176 

177 Args: 

178 plugin_name: Namespacing key this metadata was reported under (e.g. "pii_filter"). 

179 raw_metrics: The raw value at ``result.metadata[plugin_name]`` -- expected to be 

180 a ``dict[str, Any]`` but treated as untrusted (may be any type). 

181 

182 Returns: 

183 A bounded dict containing only the validated/sanitized entries. Empty if 

184 ``raw_metrics`` is not a dict or nothing survives validation. 

185 """ 

186 if not isinstance(raw_metrics, dict): 

187 logger.debug("Dropped plugin metadata for %r: value is not a dict", plugin_name) 

188 return {} 

189 

190 sanitized: Dict[str, Any] = {} 

191 # Bound the scan itself (not just the accepted count): only the first 

192 # _MAX_PLUGIN_KEYS items of the raw dict are ever inspected, so a dict 

193 # front-loaded with invalid/dropped keys can't force an unbounded scan. 

194 for key, value in itertools.islice(raw_metrics.items(), _MAX_PLUGIN_KEYS): 

195 if not _is_valid_identifier(key): 

196 logger.debug("Dropped a plugin metric key for %r: key is not a valid identifier", plugin_name) 

197 continue 

198 

199 if isinstance(value, bool): 

200 sanitized[key] = value 

201 elif isinstance(value, (int, float)): 

202 if key not in _SAFE_NUMERIC_FIELD_NAMES: 

203 logger.debug("Dropped plugin metric %r for %r: field name not in the explicit numeric allowlist", key, plugin_name) 

204 else: 

205 validated = _validate_numeric_value(value) 

206 if validated is not None: 

207 sanitized[key] = validated 

208 else: 

209 logger.debug("Dropped plugin metric %r for %r: numeric value is not finite", key, plugin_name) 

210 elif isinstance(value, str): 

211 if key not in _SAFE_STRING_FIELD_NAMES: 

212 logger.debug("Dropped plugin metric %r for %r: field name not in the explicit string allowlist", key, plugin_name) 

213 else: 

214 validated = _validate_string_value(value) 

215 if validated is not None: 

216 sanitized[key] = validated 

217 else: 

218 logger.debug("Dropped plugin metric %r for %r: string value not in safe low-cardinality contract", key, plugin_name) 

219 elif isinstance(value, list): 

220 if key not in _SAFE_STRING_FIELD_NAMES: 

221 logger.debug("Dropped plugin metric %r for %r: field name not in the explicit string allowlist", key, plugin_name) 

222 else: 

223 prefix = value[:_MAX_LIST_ITEMS] 

224 # Reject any oversized item before joining, so the join itself 

225 # (and the subsequent regex match) is always bounded by 

226 # _MAX_LIST_ITEMS * _MAX_STRING_LENGTH, not by attacker input. 

227 if all(isinstance(item, str) and len(item) <= _MAX_STRING_LENGTH for item in prefix): 

228 joined = ",".join(prefix) 

229 validated = _validate_string_value(joined) 

230 if validated is not None: 

231 sanitized[key] = validated 

232 else: 

233 logger.debug("Dropped plugin metric %r for %r: joined list value not in safe low-cardinality contract", key, plugin_name) 

234 else: 

235 logger.debug("Dropped plugin metric %r for %r: list contains non-string or oversized items", key, plugin_name) 

236 else: 

237 logger.debug("Dropped plugin metric %r for %r: type not allowed", key, plugin_name) 

238 

239 return sanitized

242def record_plugin_metrics(trace_id: Optional[str], result_metadata: Optional[Dict[str, Any]]) -> None: 

243 """Consume ``PluginResult.metadata`` from a completed ``invoke_hook`` call into observability. 

244 

245 For each ``result_metadata[<plugin>]`` entry, validates the value (S4 -- see module docstring 

246 for the exact bounds), then records the validated metrics: 

247 

248 - Primary: attaches the validated key/value pairs as attributes on a small, dedicated, 

249 short-lived span (``plugin.metrics.<plugin_name>``) rooted at the active trace. This 

250 span exists purely to carry the attributes -- it does not represent real work -- because 

251 by the time this function runs, both the hook-chain span (owned by the CPEX framework) 

252 and any per-call-site span have typically already been closed, so there is no live span 

253 left to attach attributes to in place. All spans created for a single call to this 

254 function share one DB session (P-3 batching): ``start_span``/``end_span`` both accept an 

255 external ``obs_db`` session, so the whole primary path commits once. 

256 - Secondary (optional): for each *numeric* (``int``/``float``, excluding ``bool``) validated 

257 field, also records an ``ObservabilityMetric`` row via ``record_metric()`` so plugin 

258 counters (e.g. ``total_detections``) are queryable as metrics, not just span attributes. 

259 ``record_metric()`` does not accept an external session (unlike start_span/end_span), so 

260 each call opens its own independent short-lived session -- this mirrors the existing 

261 "issue #3883" independent-session pattern used throughout ``ObservabilityService`` and is 

262 an accepted, documented limitation of batching this secondary path. 

263 

264 This is deliberately best-effort (L4): it must never raise into the request path. Any 

265 unexpected failure (missing trace, DB error, malformed plugin metadata, etc.) is logged at 

266 debug level and swallowed. No-op if ``trace_id`` or ``result_metadata`` is falsy/absent. 

267 

268 Args: 

269 trace_id: The active trace identifier (typically ``current_trace_id.get()``). If falsy, 

270 this function is a no-op (there is nothing to correlate the metrics with). 

271 result_metadata: ``PluginResult.metadata`` as returned by ``invoke_hook()`` -- expected to 

272 be a ``dict`` mapping plugin name -> that plugin's own metadata dict. Falsy/non-dict 

273 values are a no-op. 

274 """ 

275 if not trace_id or not result_metadata or not isinstance(result_metadata, dict): 

276 return 

277 

278 try: 

279 # First-Party (deferred to avoid a circular import with mcpgateway.services.__init__, 

280 # mirroring build_request_extensions() above). 

281 # First-Party 

282 from mcpgateway.config import settings # pylint: disable=import-outside-toplevel 

283 from mcpgateway.db import SessionLocal # pylint: disable=import-outside-toplevel 

284 from mcpgateway.services.observability_service import ObservabilityService # pylint: disable=import-outside-toplevel,cyclic-import 

285 

286 # Bounded iteration: cap applied via islice over the dict's own view rather than 

287 # materializing a full list of every entry first (result_metadata is untrusted). 

288 if len(result_metadata) > _MAX_PLUGINS_PER_CALL: 

289 logger.debug("Dropping %d plugin metadata entries beyond the per-call limit", len(result_metadata) - _MAX_PLUGINS_PER_CALL) 

290 plugin_items = itertools.islice(result_metadata.items(), _MAX_PLUGINS_PER_CALL) 

291 

292 sanitized_by_plugin: Dict[str, Dict[str, Any]] = {} 

293 for plugin_name, raw_metrics in plugin_items: 

294 if not _is_valid_identifier(plugin_name): 

295 logger.debug("Dropped a plugin metadata entry: plugin name is not a valid identifier") 

296 continue 

297 sanitized = _sanitize_plugin_metrics(plugin_name, raw_metrics) 

298 if sanitized: 

299 sanitized_by_plugin[plugin_name] = sanitized 

300 

301 if not sanitized_by_plugin: 

302 return 

303 

304 service = ObservabilityService() 

305 

306 # Primary: span attributes. Batch all plugins' spans into a single DB session. 

307 # Independently toggleable from the numeric-metric-row and OTel export sinks below 

308 # (DB growth, numeric-row amplification, and external-collector export are separate 

309 # operational trade-offs). 

310 if settings.plugin_metrics_db_spans_enabled: 

311 db = None 

312 try: 

313 db = SessionLocal() 

314 for plugin_name, sanitized in sanitized_by_plugin.items(): 

315 try: 

316 span_id = service.start_span( 

317 trace_id=trace_id, 

318 name=f"plugin.metrics.{plugin_name}", 

319 kind="internal", 

320 resource_type="plugin", 

321 resource_name=plugin_name, 

322 attributes=sanitized, 

323 commit=False, 

324 obs_db=db, 

325 ) 

326 service.end_span(span_id, status="ok", commit=False, obs_db=db) 

327 except Exception: # noqa: BLE001 - one plugin's failure must not affect the others 

328 logger.debug("Failed to record span attributes for plugin %r", plugin_name, exc_info=True) 

329 db.commit() 

330 except Exception: # noqa: BLE001 

331 logger.debug("Failed to batch-commit plugin metric spans", exc_info=True) 

332 if db is not None: 

333 try: 

334 db.rollback() 

335 except Exception: # nosec B110 

336 pass 

337 finally: 

338 if db is not None: 

339 try: 

340 db.close() 

341 except Exception: # nosec B110 

342 pass 

343 

344 # Secondary (optional): numeric fields also recorded as metrics. Each write opens its 

345 # own independent session (record_metric() doesn't accept an external session), so this 

346 # is additionally capped per-call (across all plugins) to bound DB write amplification, 

347 # and can be disabled entirely via settings. 

348 if settings.plugin_metrics_db_numeric_rows_enabled: 

349 numeric_rows_written = 0 

350 max_numeric = settings.plugin_metrics_max_numeric_per_call 

351 for plugin_name, sanitized in sanitized_by_plugin.items(): 

352 if numeric_rows_written >= max_numeric: 

353 break 

354 for field_name, value in sanitized.items(): 

355 if isinstance(value, bool) or not isinstance(value, (int, float)): 

356 continue 

357 if numeric_rows_written >= max_numeric: 

358 logger.debug("Dropping remaining numeric plugin metrics beyond the per-call cap of %d", max_numeric) 

359 break 

360 try: 

361 service.record_metric( 

362 name=f"plugin.{plugin_name}.{field_name}", 

363 value=float(value), 

364 metric_type="gauge", 

365 resource_type="plugin", 

366 resource_id=plugin_name, 

367 trace_id=trace_id, 

368 attributes={"plugin": plugin_name, "field": field_name}, 

369 ) 

370 numeric_rows_written += 1 

371 except Exception: # noqa: BLE001 - one metric's failure must not affect the others 

372 logger.debug("Failed to record numeric metric %r for plugin %r", field_name, plugin_name, exc_info=True) 

373 

374 # G2: optional OTel-SDK export sink. Gateway is the export authority; this 

375 # re-emits the ALREADY-sanitized metrics through the gateway's existing OTel 

376 # exporter. No-op when OTel tracing is unconfigured (create_span -> nullcontext), 

377 # and also no-op when there's no active OTel context, so a configured-but-unused 

378 # tracer doesn't turn these into orphan root spans. 

379 try: 

380 # First-Party 

381 from mcpgateway.observability import create_span, otel_context_active, otel_tracing_enabled # pylint: disable=import-outside-toplevel 

382 

383 if otel_tracing_enabled() and otel_context_active(): 

384 for plugin_name, sanitized in sanitized_by_plugin.items(): 

385 try: 

386 with create_span(f"plugin.metrics.{plugin_name}", dict(sanitized)): 

387 pass # attributes set on enter; span exported on exit 

388 except Exception: # noqa: BLE001 - one plugin's export must not affect others 

389 logger.debug("Failed to export OTel span for plugin %r", plugin_name, exc_info=True) 

390 except Exception: # noqa: BLE001 - L4: export is best-effort, never raises into the request path 

391 logger.debug("OTel plugin-metrics export sink failed", exc_info=True) 

392 except Exception: # noqa: BLE001 - L4: best-effort, must never raise into the request path 

393 logger.debug("record_plugin_metrics failed", exc_info=True)

396def build_request_extensions() -> Optional[Extensions]: 

397 """Build a CPEX ``Extensions`` object carrying the currently active trace context. 

398 

399 Reads the active trace_id/span_id from the observability ContextVars 

400 (``current_trace_id`` / ``current_span_id`` in ``mcpgateway.services.observability_service``, 

401 set by ``ObservabilityMiddleware`` for the lifetime of the request) and wraps them in a 

402 CPEX ``RequestExtension`` so plugin hook invocations (``invoke_hook(..., extensions=...)``) 

403 can correlate their execution with the originating HTTP request's trace/span. 

404 

405 This is deliberately best-effort: it must never raise into the request path. If no trace 

406 is currently active (e.g. observability is disabled, or the call happens outside of a 

407 traced request such as a background task or test), or if anything unexpected goes wrong 

408 while building the extensions, ``None`` is returned so callers can pass it straight through 

409 to ``invoke_hook(extensions=...)`` without any special-casing. 

410 

411 Note: the import of ``mcpgateway.services.observability_service`` is deliberately deferred 

412 to inside this function (rather than at module level) to avoid a circular import: that 

413 service module's package (``mcpgateway.services``) eagerly imports ``tool_service`` / 

414 ``prompt_service`` in its ``__init__.py``, both of which import this helper. 

415 

416 Returns: 

417 An ``Extensions`` instance with a populated ``request`` field when a trace_id is 

418 active, otherwise ``None``. 

419 """ 

420 try: 

421 # First-Party (deferred to avoid a circular import with mcpgateway.services.__init__) 

422 # First-Party 

423 from mcpgateway.services.observability_service import current_span_id, current_trace_id # pylint: disable=import-outside-toplevel 

424 

425 trace_id = current_trace_id.get() 

426 if not trace_id: 

427 return None 

428 span_id = current_span_id.get() 

429 return Extensions(request=RequestExtension(trace_id=trace_id, span_id=span_id)) 

430 except Exception: # noqa: BLE001 - best-effort helper, must never raise into the request path 

431 logger.debug("Failed to build CPEX request Extensions from active trace context", exc_info=True) 

432 return None

435# --------------------------------------------------------------------------- 

436# Wildcard-aware attribute policy helpers 

437# --------------------------------------------------------------------------- 

438# 

439# Rules: 

440# - Exact keys (no ``*``) take precedence over wildcard rules. 

441# - ``*`` matches exactly one dot-delimited path segment. 

442# - No arbitrary user-supplied regular expressions (no catastrophic backtracking). 

443# - Mappings into reserved ``otel.*`` names are rejected at compile time. 

444# - Empty source / destination keys are rejected at compile time. 

445# - Rules are compiled once at startup (``_compile_wildcard_rule``), not per invocation. 

446 

447# Pre-compiled sentinel: a plain [^.]+ pattern replacing one ``*`` segment. 

448_SEGMENT_PATTERN = r"[^.]+" 

449 

450 

451def _compile_wildcard_rule(src: str) -> "re.Pattern[str]": 

452 r"""Compile a single wildcard rule source key into a regex Pattern. 

453 

454 Only ``*`` is treated specially — it maps to ``[^.]+`` (one dot-delimited 

455 segment). All other regex metacharacters in the source key are escaped, so 

456 there is no possibility of catastrophic backtracking from user-supplied 

457 patterns. 

458 

459 Args: 

460 src: Source attribute key with zero or more ``*`` segments 

461 (e.g. ``"cpex.control.results.*.result.reason"``). 

462 

463 Returns: 

464 A compiled regex that matches the full attribute key when the ``*`` 

465 positions contain exactly one non-dot segment. 

466 

467 Example: 

468 >>> import re 

469 >>> p = _compile_wildcard_rule("cpex.control.results.*.result.reason") 

470 >>> bool(p.fullmatch("cpex.control.results.pii-guard.result.reason")) 

471 True 

472 >>> bool(p.fullmatch("cpex.control.results.a.b.result.reason")) 

473 False 

474 """ 

475 # Split on literal ``*``, escape surrounding parts, join with segment pattern. 

476 parts = src.split("*") 

477 regex = _SEGMENT_PATTERN.join(re.escape(p) for p in parts) 

478 return re.compile(r"\A" + regex + r"\Z")

481def compile_attribute_policy( 

482 mapping: Dict[str, str], 

483 removals: List[str], 

484) -> "Tuple[List[Tuple[bool, Any, str, str]], List[Tuple[bool, Any, str]]]": 

485 """Compile attribute mapping and removal rules for efficient per-invocation use. 

486 

487 Rules are compiled **once** at startup (e.g. inside a plugin ``__init__``). 

488 Calling this per-invocation is unnecessary and wasteful. 

489 

490 Rule precedence: 

491 - Exact rules (no ``*``) take precedence over wildcard rules. 

492 - Among wildcard rules, the first matching rule wins. 

493 - Removal is evaluated after rename. 

494 

495 Safety: 

496 - ``otel.*`` destination names are rejected. 

497 - Empty source or destination keys are rejected. 

498 - No user-supplied regular expressions (``*`` only). 

499 

500 Args: 

501 mapping: Rename rules ``{source_key: dest_key}``. 

502 removals: List of source keys to drop from attributes. 

503 

504 Returns: 

505 A 2-tuple ``(compiled_mappings, compiled_removals)``. 

506 

507 Each mapping entry is a 4-tuple 

508 ``(is_exact: bool, pattern: Optional[re.Pattern], src: str, dst: str)``. 

509 

510 Each removal entry is a 3-tuple 

511 ``(is_exact: bool, pattern: Optional[re.Pattern], src: str)``. 

512 

513 Raises: 

514 ValueError: On empty keys, reserved destinations, or invalid patterns. 

515 

516 Example: 

517 >>> mappings, removals = compile_attribute_policy( 

518 ... {"cpex.control.result.allowed": "controls.result.allow"}, 

519 ... ["cpex.control.config.keys"], 

520 ... ) 

521 >>> len(mappings) 

522 1 

523 >>> len(removals) 

524 1 

525 """ 

526 compiled_mappings: List[Tuple[bool, Any, str, str]] = [] 

527 compiled_removals: List[Tuple[bool, Any, str]] = [] 

528 

529 for src, dst in mapping.items(): 

530 if not src or not dst: 

531 raise ValueError(f"Attribute mapping keys must be non-empty: {src!r} -> {dst!r}") 

532 if dst.startswith("otel."): 

533 raise ValueError(f"Mapping into reserved 'otel.*' namespace is not allowed: {dst!r}") 

534 if len(src) > 256 or len(dst) > 256: 

535 raise ValueError(f"Attribute mapping key exceeds 256 characters: {src!r} -> {dst!r}") 

536 if "*" in src: 

537 pattern = _compile_wildcard_rule(src) 

538 compiled_mappings.append((False, pattern, src, dst)) 

539 else: 

540 compiled_mappings.append((True, None, src, dst)) 

541 

542 for src in removals: 

543 if not src: 

544 continue 

545 if len(src) > 256: 

546 raise ValueError(f"Remove-attribute key exceeds 256 characters: {src!r}") 

547 if "*" in src: 

548 pattern = _compile_wildcard_rule(src) 

549 compiled_removals.append((False, pattern, src)) 

550 else: 

551 compiled_removals.append((True, None, src)) 

552 

553 return compiled_mappings, compiled_removals

556def _apply_one_mapping_rename( 

557 key: str, 

558 compiled_mappings: "List[Tuple[bool, Any, str, str]]", 

559) -> str: 

560 """Return the renamed key after applying compiled mapping rules. 

561 

562 Exact rules take precedence over wildcard rules. The first matching 

563 wildcard rule wins. If no rule matches, the key is returned unchanged. 

564 

565 ``*`` captured segments are substituted into the destination template at 

566 the same position. 

567 

568 Args: 

569 key: Original attribute key. 

570 compiled_mappings: Output of ``compile_attribute_policy``. 

571 

572 Returns: 

573 Renamed key, or the original key when no rule matches. 

574 """ 

575 # Pass 1: exact rules (is_exact=True) 

576 for is_exact, _pattern, src, dst in compiled_mappings: 

577 if is_exact and src == key: 

578 return dst 

579 

580 # Pass 2: wildcard rules (is_exact=False) 

581 for is_exact, pattern, src, dst in compiled_mappings: 

582 if is_exact: 

583 continue 

584 m = pattern.fullmatch(key) if pattern else None 

585 if m: 

586 # Reconstruct destination by replacing ``*`` placeholders with the 

587 # captured segments. The number of ``*`` in src and dst must match; 

588 # if they differ, skip silently (misconfiguration — already warned at 

589 # compile time ideally; here we are defensive). 

590 src_parts = src.split("*") 

591 dst_parts = dst.split("*") 

592 if len(src_parts) != len(dst_parts): 

593 continue 

594 # Extract captured segments: they are the substrings of key that 

595 # correspond to the ``*`` slots. 

596 key_remainder = key 

597 captured: List[str] = [] 

598 for i, prefix in enumerate(src_parts[:-1]): 

599 if not key_remainder.startswith(re.escape(prefix)): 

600 # Safety: let the regex result guide us but strip literal prefixes 

601 idx = len(prefix) 

602 else: 

603 idx = len(prefix) 

604 key_remainder = key_remainder[idx:] 

605 # Find the end of the captured segment (next ``re.escape(src_parts[i+1])``) 

606 suffix = src_parts[i + 1] 

607 if suffix: 

608 cap_end = key_remainder.find(suffix) 

609 if cap_end < 0: 

610 break 

611 captured.append(key_remainder[:cap_end]) 

612 key_remainder = key_remainder[cap_end:] 

613 else: 

614 captured.append(key_remainder) 

615 key_remainder = "" 

616 else: 

617 # Build destination by interleaving dst_parts and captured 

618 if len(captured) == len(dst_parts) - 1: 

619 result = dst_parts[0] 

620 for seg, dpart in zip(captured, dst_parts[1:]): 

621 result += seg + dpart 

622 return result 

623 return key

626def apply_attribute_mapping(attributes: dict, mapping: dict) -> dict: 

627 """Apply attribute name mapping (renaming) to a dictionary of attributes. 

628 

629 Supports both exact key renames and single-``*`` wildcard renames. 

630 Exact rules take precedence over wildcard rules. 

631 

632 Args: 

633 attributes: Dictionary of attributes to rename. 

634 mapping: Dictionary mapping old attribute names to new names. 

635 Wildcard rules use ``*`` to match one dot-delimited segment. 

636 

637 Returns: 

638 New dictionary with renamed attributes. 

639 

640 Example: 

641 >>> attrs = {"tool.name": "weather", "tool.version": "1.0"} 

642 >>> mapping = {"tool.name": "controls.artifact.name"} 

643 >>> apply_attribute_mapping(attrs, mapping) 

644 {'controls.artifact.name': 'weather', 'tool.version': '1.0'} 

645 

646 >>> attrs2 = {"cpex.control.results.pii.result.allowed": True} 

647 >>> mapping2 = {"cpex.control.results.*.result.allowed": "controls.results.*.result.allowed"} 

648 >>> result2 = apply_attribute_mapping(attrs2, mapping2) 

649 >>> "controls.results.pii.result.allowed" in result2 

650 True 

651 """ 

652 if not mapping: 

653 return dict(attributes) 

654 

655 try: 

656 compiled_mappings, _ = compile_attribute_policy(mapping, []) 

657 except ValueError: 

658 # Misconfigured mapping at runtime — fall back to exact-only behaviour 

659 logger.debug("apply_attribute_mapping: compile_attribute_policy failed; falling back to exact match", exc_info=True) 

660 compiled_mappings = [(True, None, src, dst) for src, dst in mapping.items()] 

661 

662 renamed_attributes: dict = {} 

663 for old_name, value in attributes.items(): 

664 new_name = _apply_one_mapping_rename(old_name, compiled_mappings) 

665 renamed_attributes[new_name] = value 

666 

667 logger.debug("Applied %d attribute name mappings", len(mapping)) 

668 return renamed_attributes
